class MonitorApp(App):
    def build(self):

        self.running = False
        self.log_text = ""
        
        layout = BoxLayout(orientation='vertical', padding=20, spacing=10)

        btn_layout.add_widget(self.start_btn)

    def toggle_monitor(self, instance):
        if not self.running:
            self.start_monitor()
        else:
            self.stop_monitor()
    
    def start_monitor(self):
        self.running = True
        self.start_btn.text = "監察中... (按停止)"
        self.status_label.text = "狀態: 🔥 監察中 - apple.com/hk"
        self.thread = threading.Thread(target=self.run_loop, daemon=True)
        self.thread.start()
        self.log("🔥 監察已啟動 - 目標 apple.com/hk")
    
    def stop_monitor(self):
        self.running = False
        self.start_btn.text = "🔥 開始監察"
        self.status_label.text = "狀態: 已停止"
        self.log("⏸️ 已停止")
    
    def log(self, msg):
        t = datetime.now().strftime("%H:%M:%S")
        self.log_text = f"[{t}] {msg}\n" + self.log_text[:500]
        Clock.schedule_once(lambda dt: setattr(self.log_label, 'text', self.log_text))
    
    def run_loop(self):
        asyncio.run(self.async_loop())
    
    async def async_loop(self):
        async with aiohttp.ClientSession() as session:
            await self.send_telegram("🔥 *監察已啟動* ✅\n🎯 apple.com/hk\n📱 8 SKU\n⭐ 優先 布地根紅色 512GB")
            last = {sku: False for sku in SKU_MAP}
            while self.running:
                try:
                    params = {"pl": "true", "parts": ",".join(SKU_MAP.keys()), "searchNearby": "true", "store": "R428"}
                    async with session.get(API, params=params, headers={"User-Agent": "Mozilla/5.0 (iPhone)"}, timeout=10) as r:
                        if r.status == 200:
                            data = await r.json()
                            stores = data.get("body",{}).get("content",{}).get("pickupMessage",{}).get("stores",[])
                            for sku in SKU_MAP:
                                av = False
                                for st in stores:
                                    pa = st.get("partsAvailability",{}).get(sku,{})
                                    if pa.get("pickupDisplay")=="available" or pa.get("buyability",{}).get("isBuyable"):
                                        av = True; break
                                info = SKU_MAP[sku]
                                if av and not last[sku]:
                                    last[sku]=True
                                    await self.send_telegram(f"🚨 *開售啦！*\n\n{info['emoji']} {info['color']} {info['storage']}\n🔗 https://www.apple.com/hk/shop/buy-iphone/iphone-18-pro")
                                    self.log(f"🚨 {info['color']} {info['storage']} 開售！")
                                elif not av:
                                    last[sku]=False
                                    self.log(f"{info['emoji']} {info['color']} 監察中...")
                except Exception as e:
                    self.log(f"錯誤: {e}")
                await asyncio.sleep(5)

    async def send_telegram(self, text):
        url = f"https://api.telegram.org/bot{DEFAULT_TOKEN}/sendMessage"
        payload = {"chat_id": DEFAULT_CHAT, "text": text, "parse_mode": "Markdown"}
        async with aiohttp.ClientSession() as s:
            try: await s.post(url, json=payload, timeout=10)
            except: pass
